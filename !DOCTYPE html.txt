let current = 0;
let answers = [];

const app = document.getElementById('app');

function saveProgress() {
  localStorage.setItem('stardew_quiz_answers', JSON.stringify(answers));
  localStorage.setItem('stardew_quiz_current', current);
}

function loadProgress() {
  const saved = localStorage.getItem('stardew_quiz_answers');
  if (saved) {
    answers = JSON.parse(saved);
    current = parseInt(localStorage.getItem('stardew_quiz_current') || 0);
  }
}

function showStart() {
  app.innerHTML = `
    <div class="start-screen">
      <h1>Привет!</h1>
      <p>Это маленький опрос в стиле Stardew Valley.<br>
      Ответь честно на вопросы про свои предпочтения.<br>
      В конце сможешь скопировать все ответы и отправить.</p>
      <button onclick="startQuiz()">Начать ❤</button>
    </div>
  `;
}

function startQuiz() {
  current = 0;
  answers = [];
  localStorage.removeItem('stardew_quiz_answers');
  localStorage.removeItem('stardew_quiz_current');
  showQuestion();
}

function showQuestion() {
  if (current >= questions.length) {
    showResults();
    return;
  }
  
  const q = questions[current];
  const progress = ((current) / questions.length) * 100;
  
  app.innerHTML = `
    <div class="progress"><div class="progress-bar" style="width:${progress}%"></div></div>
    <div class="content">
      <div class="portrait">👩‍🦱</div>
      <div class="dialogue">
        <div class="question">${q.q}</div>
        <div class="options">
          ${q.options.map((opt, i) => 
            `<div class="option" onclick="selectOption(${i})">${opt}</div>`
          ).join('')}
        </div>
      </div>
    </div>
    <div class="nav">
      <button onclick="prevQuestion()" ${current === 0 ? 'disabled' : ''}>← Назад</button>
      <span style="font-size:10px;align-self:center">${current + 1} / ${questions.length}</span>
      <button id="nextBtn" onclick="nextQuestion()" disabled>Далее →</button>
    </div>
  `;
}

let selected = null;

function selectOption(i) {
  selected = i;
  document.querySelectorAll('.option').forEach((el, idx) => {
    el.classList.toggle('selected', idx === i);
  });
  document.getElementById('nextBtn').disabled = false;
}

function nextQuestion() {
  if (selected === null) return;
  answers[current] = {
    question: questions[current].q,
    answer: questions[current].options[selected]
  };
  selected = null;
  current++;
  saveProgress();
  showQuestion();
}

function prevQuestion() {
  if (current > 0) {
    current--;
    selected = null;
    showQuestion();
  }
}

function showResults() {
  const text = answers.map((a, i) => 
    `${i + 1}. ${a.question}\n→ ${a.answer}`
  ).join('\n\n');
  
  app.innerHTML = `
    <div class="results">
      <h2>❤ Результаты опроса ❤</h2>
      ${answers.map((a, i) => `
        <div class="result-item">
          <strong>${i + 1}. ${a.question}</strong><br>
          → ${a.answer}
        </div>
      `).join('')}
      <button class="copy-btn" onclick="copyResults()">📋 Скопировать все ответы</button>
      <button style="margin-top:12px;width:100%;background:#FFCC80;border-color:#EF6C00" onclick="startQuiz()">Пройти заново</button>
    </div>
  `;
  
  // сохраняем финальные ответы
  localStorage.setItem('stardew_quiz_final', text);
}

function copyResults() {
  const text = `Результаты опроса предпочтений (Stardew Style):\n\n` + 
    answers.map((a, i) => `${i + 1}. ${a.question}\n→ ${a.answer}`).join('\n\n');
  
  navigator.clipboard.writeText(text).then(() => {
    alert('Результаты скопированы! Теперь можешь вставить их в сообщение 💕');
  }).catch(() => {
    // fallback
    prompt('Скопируй текст вручную:', text);
  });
}

// запуск
loadProgress();
if (answers.length === questions.length) {
  showResults();
} else if (answers.length > 0) {
  showQuestion();
} else {
  showStart();
}
